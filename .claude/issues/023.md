## タスク概要

ccskill-spreadsheet の `lib/registry.sh` を移植し、Gmail 用に調整する。
全インストール先をグローバルに追跡する `.registry.json` の CRUD 操作を提供。

## 背景

現状は各プロジェクトの `.ccskill-metadata.json` に個別記録のみで、
「どのプロジェクトにインストール済みか」を一覧する手段がない。
レジストリにより `status` / `update-all` コマンドが実現可能になる。

## 作成ファイル

### `lib/registry.sh`

```bash
#!/bin/bash
#
# registry.sh - Installation registry helper functions
#
# Usage:
#   source "$CCSKILL_GMAIL_DIR/lib/registry.sh"
#   registry_upsert "$TARGET_DIR"
#
# Registry file: $CCSKILL_GMAIL_DIR/.registry.json
#

# Registry file path
_REGISTRY_FILE="${CCSKILL_GMAIL_DIR:-.}/.registry.json"

_REGISTRY_JQ_WARNED=false

_registry_has_jq() {
    if command -v jq &> /dev/null; then return 0; fi
    if [ "$_REGISTRY_JQ_WARNED" = false ]; then
        _REGISTRY_JQ_WARNED=true
        echo -e "\033[1;33mNote: jq is not installed - registry features disabled\033[0m" >&2
    fi
    return 1
}

_registry_now() { date '+%Y-%m-%d %H:%M:%S'; }

_registry_write() {
    local tmp; tmp=$(mktemp)
    echo "$1" > "$tmp"
    /bin/mv -f "$tmp" "$_REGISTRY_FILE"
}
```

## spreadsheet 版との差分

| 項目 | spreadsheet | gmail |
|------|-------------|-------|
| `_REGISTRY_FILE` | `$CCSKILL_SPREADSHEET_DIR/.registry.json` | `$CCSKILL_GMAIL_DIR/.registry.json` |
| `registry_upsert` メタデータパス | `.ccskill-spreadsheet/.ccskill-metadata.json` | `.ccskill-gmail/.ccskill-metadata.json` |
| `registry_upsert` フィールド | `spreadsheets` | なし（Gmail はマルチ概念不要） |
| `registry_verify` チェック先 | `.ccskill-spreadsheet` | `.ccskill-gmail` |

### registry_upsert の Gmail 版

```bash
registry_upsert() {
    _registry_has_jq || return 0

    local target_dir="$1"
    if [ -z "$target_dir" ]; then return 1; fi

    if [ -d "$target_dir" ]; then
        target_dir=$(cd "$target_dir" && pwd)
    fi

    registry_init

    local metadata_file="$target_dir/.ccskill-gmail/.ccskill-metadata.json"
    local now; now=$(_registry_now)

    local project_name installed_at version

    if [ -f "$metadata_file" ]; then
        project_name=$(jq -r '.project_name // empty' "$metadata_file" 2>/dev/null)
        installed_at=$(jq -r '.installed_at // empty' "$metadata_file" 2>/dev/null)
        version=$(jq -r '.version // "unknown"' "$metadata_file" 2>/dev/null)
    else
        project_name=$(basename "$target_dir")
        installed_at="$now"
        version="unknown"
    fi

    # Gmail版: spreadsheets フィールド不要
    local existing_installed_at
    existing_installed_at=$(jq -r --arg path "$target_dir" \
        '.installations[$path].installed_at // empty' "$_REGISTRY_FILE" 2>/dev/null)
    if [ -n "$existing_installed_at" ]; then
        installed_at="$existing_installed_at"
    fi

    local tmp; tmp=$(mktemp)
    jq --arg path "$target_dir" \
       --arg project_name "${project_name:-$(basename "$target_dir")}" \
       --arg installed_at "${installed_at:-$now}" \
       --arg updated_at "$now" \
       --arg version "${version:-unknown}" \
       '.installations[$path] = {
          project_name: $project_name,
          installed_at: $installed_at,
          updated_at: $updated_at,
          version: $version
        } | .updated_at = $updated_at' \
       "$_REGISTRY_FILE" > "$tmp"
    /bin/mv -f "$tmp" "$_REGISTRY_FILE"
}
```

### registry_verify の Gmail 版

```bash
registry_verify() {
    local target_dir="$1"
    if [ -z "$target_dir" ]; then return 1; fi
    if [ ! -d "$target_dir" ]; then return 1; fi
    if [ ! -d "$target_dir/.ccskill-gmail" ]; then return 2; fi
    return 0
}
```
