|-----|------|------|
| search | GET | メール検索（Gmail 検索構文） |

## 作成するファイル

### handlers/search.js

```javascript
// search: Gmail検索クエリでメールを検索
// パラメータ: query (必須), maxResults (オプション, デフォルト20)
function handleSearch(query, maxResults) {
  const threads = GmailApp.search(query, 0, maxResults || 20);
  return threads.map(formatThread);
}
```

### handlers/draft.js

```javascript
// create_draft: 新規下書き作成
// パラメータ: to, subject, body, cc (オプション), bcc (オプション)
function handleCreateDraft(to, subject, body, cc, bcc) {
  const draft = GmailApp.createDraft(to, subject, body, {
    cc: cc,
    bcc: bcc
  });
  return {
    draftId: draft.getId(),
    message: '下書きを作成しました',
    gmailUrl: 'https://mail.google.com/mail/u/0/#drafts'
  };
}
```
