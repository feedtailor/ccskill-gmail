## 1. commands/register.sh

### 設計

```bash
#!/bin/bash
set -e

if [ -z "$CCSKILL_GMAIL_DIR" ]; then
    echo "Error: This script should be called via 'ccskill-gmail register'"
    exit 1
fi

source "$CCSKILL_GMAIL_DIR/lib/registry.sh"

if [ $# -eq 0 ]; then
    echo "Usage: ccskill-gmail register <PATH> [PATH...]"
    exit 1
fi

REGISTERED=0
SKIPPED=0

for path in "$@"; do
    # 絶対パスに変換
    if [ -d "$path" ]; then
        path=$(cd "$path" && pwd)
    else
        echo "  ✗ $path - ディレクトリが存在しません"
        SKIPPED=$((SKIPPED + 1))
        continue
    fi

    # インストール存在チェック
    if [ ! -d "$path/.ccskill-gmail" ]; then
        echo "  ✗ $path - .ccskill-gmail/ が見つかりません"
        SKIPPED=$((SKIPPED + 1))
        continue
    fi

    registry_upsert "$path"
    echo "  ✓ $path"
    REGISTERED=$((REGISTERED + 1))
done

echo ""
echo "Registered: $REGISTERED, Skipped: $SKIPPED"
```

## 2. commands/apply-config.sh

### 設計

Gmail の `config.js` は現状ほぼ空だが、将来拡張される可能性あり。
config.js を変更した後、GAS に反映するためのコマンド。

```bash
#!/bin/bash
set -e

if [ -z "$CCSKILL_GMAIL_DIR" ]; then
    echo "Error: This script should be called via 'ccskill-gmail apply-config'"
    exit 1
fi

source "$CCSKILL_GMAIL_DIR/lib/push-gas.sh"

TARGET_DIR="${1:-.}"
TARGET_DIR=$(cd "$TARGET_DIR" && pwd)

GAS_DIR="$TARGET_DIR/.ccskill-gmail"

# 前提チェック
if [ ! -d "$GAS_DIR" ]; then
    echo "Error: .ccskill-gmail/ が見つかりません。"
    exit 1
fi

if [ ! -f "$GAS_DIR/config.js" ]; then
    echo "Error: config.js が見つかりません。"
    exit 1
fi

echo "config.js の変更を GAS に反映します..."

# push
push_gas "$GAS_DIR" "$CCSKILL_GMAIL_DIR"
echo "✓ GAS コード push 完了"

# 自動デプロイ
METADATA_FILE="$GAS_DIR/.ccskill-metadata.json"
DEPLOYMENT_ID=""
if [ -f "$METADATA_FILE" ]; then
    DEPLOYMENT_ID=$(jq -r '.deployment_id // empty' "$METADATA_FILE" 2>/dev/null)
fi

if [ -n "$DEPLOYMENT_ID" ]; then
    deploy_gas "$GAS_DIR" "$DEPLOYMENT_ID" "Config update"
    echo "✓ 自動デプロイ完了"
else
    echo ""
    echo "⚠ deployment_id が未設定のため、手動デプロイが必要です。"
    SCRIPT_ID=$(jq -r '.scriptId' "$GAS_DIR/.clasp.json" 2>/dev/null)
    if [ -n "$SCRIPT_ID" ]; then
        echo "  https://script.google.com/d/${SCRIPT_ID}/edit"
    fi
fi
```
