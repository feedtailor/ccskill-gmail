## 主要変更点の詳細

### A. ディスパッチャ経由の呼び出し

```bash
#!/bin/bash
# commands/install.sh - ccskill-gmail install command
set -e

# CCSKILL_GMAIL_DIR is set by the dispatcher
if [ -z "$CCSKILL_GMAIL_DIR" ]; then
    echo "Error: This script should be called via 'ccskill-gmail install'"
    exit 1
fi

# Load helpers
source "$CCSKILL_GMAIL_DIR/lib/push-gas.sh"
source "$CCSKILL_GMAIL_DIR/lib/registry.sh"
source "$CCSKILL_GMAIL_DIR/lib/permissions.sh"
```

### B. マスターアーキテクチャ（src/ 廃止）

```bash
# Before: install.sh がフルコピー
/bin/cp -r "$CCSKILL_GMAIL_DIR/gas-template/"* "$GAS_DIR/src/"

# After: config.js + .clasp.json + auth.sh + api.sh のみ
GAS_DIR="$TARGET_DIR/.ccskill-gmail"
mkdir -p "$GAS_DIR"

# config.js 生成
/bin/cp "$CCSKILL_GMAIL_DIR/gas-template/config.js.template" "$GAS_DIR/config.js"

# auth.sh / api.sh コピー（SKILL から source するため）
/bin/cp "$CCSKILL_GMAIL_DIR/lib/auth.sh" "$GAS_DIR/auth.sh"
/bin/cp "$CCSKILL_GMAIL_DIR/lib/api.sh" "$GAS_DIR/api.sh"
```

### C. clasp create + 自動デプロイ

```bash
# GAS プロジェクト作成
(cd "$GAS_DIR" && clasp create --type standalone --title "Gmail Skill - $PROJECT_NAME")

# .clasp.json の rootDir を調整（push_gas が "." を使うため不要だが念のため）
if command -v jq &>/dev/null; then
    jq '.rootDir = "."' "$GAS_DIR/.clasp.json" > "$GAS_DIR/.clasp.json.tmp"
    /bin/mv "$GAS_DIR/.clasp.json.tmp" "$GAS_DIR/.clasp.json"
fi

# マスターから push
push_gas "$GAS_DIR" "$CCSKILL_GMAIL_DIR"

# 自動デプロイ
DEPLOY_RESULT=$(mktemp)
deploy_gas "$GAS_DIR" "" "Initial deployment" "$DEPLOY_RESULT"
DEPLOYMENT_ID=$(cat "$DEPLOY_RESULT")
rm -f "$DEPLOY_RESULT"
```

### D. エンドポイント検証（Bearer トークン付き）

```bash
# auth.sh を source してトークン取得
source "$CCSKILL_GMAIL_DIR/lib/auth.sh"

# GAS Script ID からエンドポイント URL を構築
SCRIPT_ID=$(jq -r '.scriptId' "$GAS_DIR/.clasp.json")
GMAIL_ENDPOINT="https://script.google.com/macros/s/${DEPLOYMENT_ID}/exec"

# ヘルスチェック
echo "エンドポイントを検証中..."
RESPONSE=$(curl -sL --max-time 60 \
    -H "Authorization: Bearer $(gas_token)" \
    "$GMAIL_ENDPOINT" 2>/dev/null)

if echo "$RESPONSE" | jq -e '.ok == true' > /dev/null 2>&1; then
    echo "✓ エンドポイント検証成功"
else
    echo "⚠ エンドポイント検証失敗"
    echo "レスポンス: $RESPONSE"
    echo ""
    echo "初回デプロイ時は GAS の OAuth 承認が必要な場合があります。"
    echo "GAS エディタを開いて手動で一度実行してください:"
    # clasp open-script で GAS エディタを開く
fi
```

### E. メタデータ保存

```bash
# .ccskill-metadata.json
cat > "$GAS_DIR/.ccskill-metadata.json" << EOF
{
  "installed_at": "$(date '+%Y-%m-%d %H:%M:%S')",
  "updated_at": "$(date '+%Y-%m-%d %H:%M:%S')",
  "installed_from": "$CCSKILL_GMAIL_DIR",
  "version": "$VERSION",
  "architecture": "master",
  "project_name": "$PROJECT_NAME",
  "deployment_id": "$DEPLOYMENT_ID"
}
EOF
```

### F. .env 保存

```bash
# .env に GMAIL_ENDPOINT を追加/更新
if [ -f "$TARGET_DIR/.env" ]; then
    # 既存エントリを削除
    tmp=$(mktemp)
    awk '!/^GMAIL_ENDPOINT=/' "$TARGET_DIR/.env" > "$tmp"
    /bin/mv "$tmp" "$TARGET_DIR/.env"
fi
echo "GMAIL_ENDPOINT=$GMAIL_ENDPOINT" >> "$TARGET_DIR/.env"
```
