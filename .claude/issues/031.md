## タスク概要

レジストリに登録された全インストール先を一括更新する `commands/update-all.sh` を新規作成。

## 設計

```bash
#!/bin/bash
set -e

if [ -z "$CCSKILL_GMAIL_DIR" ]; then
    echo "Error: This script should be called via 'ccskill-gmail update-all'"
    exit 1
fi

source "$CCSKILL_GMAIL_DIR/lib/registry.sh"

MASTER_VERSION=$(cd "$CCSKILL_GMAIL_DIR" && git rev-parse --short HEAD 2>/dev/null || echo "unknown")

registry_init

# 更新対象を特定
PATHS=$(jq -r '.installations | keys[]' "$_REGISTRY_FILE" 2>/dev/null)

if [ -z "$PATHS" ]; then
    echo "No installations registered."
    exit 0
fi

TARGETS=()
SKIPPED=0
NOT_FOUND=0

for path in $PATHS; do
    if [ ! -d "$path" ]; then
        NOT_FOUND=$((NOT_FOUND + 1))
        continue
    fi

    version=$(jq -r --arg p "$path" '.installations[$p].version // "unknown"' "$_REGISTRY_FILE")
    if [ "$version" != "$MASTER_VERSION" ]; then
        TARGETS+=("$path")
    else
        SKIPPED=$((SKIPPED + 1))
    fi
done

echo "ccskill-gmail update-all (target version: $MASTER_VERSION)"
echo ""
echo "  Up to date: $SKIPPED"
echo "  Outdated:   ${#TARGETS[@]}"
echo "  Not found:  $NOT_FOUND"

if [ ${#TARGETS[@]} -eq 0 ]; then
    echo ""
    echo "All installations are up to date."
    exit 0
fi

echo ""
echo "以下のインストール先を更新します："
for target in "${TARGETS[@]}"; do
    version=$(jq -r --arg p "$target" '.installations[$p].version // "unknown"' "$_REGISTRY_FILE")
    echo "  $target ($version → $MASTER_VERSION)"
done

echo ""
read -p "更新を実行しますか？ (y/N): " -n 1 -r
echo
if [[ ! $REPLY =~ ^[Yy]$ ]]; then
    echo "キャンセルしました。"
    exit 0
fi

# 一括更新実行
SUCCESS=0
FAILED=0
MANUAL_DEPLOY_NEEDED=()

for target in "${TARGETS[@]}"; do
    echo ""
    echo "━━━ Updating: $target ━━━"
    if "$CCSKILL_GMAIL_DIR/commands/update.sh" --yes "$target"; then
        SUCCESS=$((SUCCESS + 1))

        # deployment_id がない場合は手動デプロイ必要
        metadata="$target/.ccskill-gmail/.ccskill-metadata.json"
        deploy_id=$(jq -r '.deployment_id // empty' "$metadata" 2>/dev/null)
        if [ -z "$deploy_id" ]; then
            MANUAL_DEPLOY_NEEDED+=("$target")
        fi
    else
        FAILED=$((FAILED + 1))
    fi
done

# サマリー
echo ""
echo "━━━ 結果 ━━━"
echo "  成功: $SUCCESS"
echo "  失敗: $FAILED"

if [ ${#MANUAL_DEPLOY_NEEDED[@]} -gt 0 ]; then
    echo ""
    echo "⚠ 以下のプロジェクトは手動デプロイが必要です："
    for target in "${MANUAL_DEPLOY_NEEDED[@]}"; do
        echo "  $target"
    done
fi
```
